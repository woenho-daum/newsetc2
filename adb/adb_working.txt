scrcpy-tcp.vbs
{
	Dim WshShell, strCommand, targetIP, targetPort
	Set WshShell = CreateObject("Wscript.Shell")

	' 사용할 안드로이드 기기의 고정 IP와 ADB 포트
	targetIP = "192.168.137.115"
	targetPort = "5555"

	' 1. adb connect 명령을 먼저 실행 (연결이 완료될 때까지 대기: true)
	WshShell.Run "cmd /c adb connect " & targetIP & ":" & targetPort, 0, true

	' 2. 기존 scrcpy 실행 명령어 조합
	strCommand = "cmd /c scrcpy.exe -K --window-width=432"

	For Each Arg In WScript.Arguments
		strCommand = strCommand & " """ & replace(Arg, """", """""""""") & """"
	Next

	' 3. scrcpy 실행 (비동기 실행: false)
	WshShell.Run strCommand, 0, false
}
