SELECT sqlite_version();

SELECT
    ds.*
FROM dispatch_settings ds
LEFT JOIN contacts c
    ON c.fn_front = ds.driver_name
WHERE c.fn_front IS NULL;

UPDATE contacts
SET
    fn_new = ds.driver_name || ' ' || ds.route || ' ' || substr(ds.car_number, -4),
    update_date = CURRENT_TIMESTAMP,
    fn_order = '고정/쉬프트'
FROM dispatch_settings ds
WHERE ds.driver_name = contacts.fn_front;

SELECT
    c.fn_front,
    c.fn_old,
    c.tel_number,
    c.tel_section
FROM contacts c
LEFT JOIN dispatch_settings ds
    ON c.fn_front = ds.driver_name
WHERE ds.driver_name IS NULL;
